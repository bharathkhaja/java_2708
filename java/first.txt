// Method 1: Extending Thread class

class GoodMorningThread extends Thread {
    public void run() {
        while (true) {
            System.out.println("Good Morning");
            try {
                Thread.sleep(1000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}

class HelloThread extends Thread {
    public void run() {
        while (true) {
            System.out.println("Hello");
            try {
                Thread.sleep(2000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}

class WelcomeThread extends Thread {
    public void run() {
        while (true) {
            System.out.println("Welcome");
            try {
                Thread.sleep(3000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}


// Method 2: Implementing Runnable interface

class GoodMorningRunnable implements Runnable {
    public void run() {
        while (true) {
            System.out.println("Good Morning");
            try {
                Thread.sleep(1000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}

class HelloRunnable implements Runnable {
    public void run() {
        while (true) {
            System.out.println("Hello");
            try {
                Thread.sleep(2000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}

class WelcomeRunnable implements Runnable {
    public void run() {
        while (true) {
            System.out.println("Welcome");
            try {
                Thread.sleep(3000);
            } catch (InterruptedException e) {
                break;
            }
        }
    }
}


public class Main {
    public static void main(String[] args) {

        // Method 1: Extending Thread
        Thread t1 = new GoodMorningThread();
        Thread t2 = new HelloThread();
        Thread t3 = new WelcomeThread();

        t1.start();
        t2.start();
        t3.start();

        // Method 2: Implementing Runnable
        Thread r1 = new Thread(new GoodMorningRunnable());
        Thread r2 = new Thread(new HelloRunnable());
        Thread r3 = new Thread(new WelcomeRunnable());

        r1.start();
        r2.start();
        r3.start();
    }
}
